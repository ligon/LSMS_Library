#+begin_src sh :eval never-export :tangle no
mkdir -p "$UGA_DATA" "$UGA_OUTPUT"

  --out "$UGA_OUTPUT" --tag "$UGA_TAG" --expected-source "$UGA_SOURCE" \
  > "$UGA_OUTPUT/$UGA_TAG-build.log" 2>&1

* Compare and attribute

Set =UGA_BASE_OUTPUT= and =UGA_FINAL_OUTPUT= to the saved-frame directories above; set =UGA_ATTR_WORK= to a private scratch directory for copies of versioned non-microdata inputs.
Run from the final checkout with the same import/config environment.

#+begin_src sh :eval never-export :tangle no

  --baseline-dir "$UGA_BASE_OUTPUT" --baseline-tag base \

  --report "$UGA_FINAL_OUTPUT/comparison.json"
