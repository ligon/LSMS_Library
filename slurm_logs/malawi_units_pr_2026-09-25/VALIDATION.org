#+begin_src python :eval never-export :tangle no :exports code
import json
import os
from pathlib import Path
import warnings
import pandas as pd
import lsms_library as ll
from lsms_library.diagnostics import load_feature

assert Path(ll.__file__).resolve().parent == Path.cwd() / 'lsms_library'
output = Path(os.environ['MALAWI_OUTPUT'])
output.mkdir(parents=True, exist_ok=True)
assert not list((Path(os.environ['LSMS_DATA_DIR']) / 'Malawi').rglob('*.parquet'))
country = ll.Country('Malawi')
first = ['panel_ids', 'sample', 'cluster_features']
last = ['food_expenditures', 'food_prices', 'food_quantities',
        'household_characteristics']
order = first + [n for n in country.data_scheme if n not in first + last] + last
results = []
for name in order:
    with warnings.catch_warnings(record=True) as caught:
        warnings.simplefilter('always')
        frame = load_feature(country, name)
    if isinstance(frame, pd.DataFrame):
        frame.to_pickle(output / (name + '.pkl'))
    results.append(dict(table=name, rows=len(frame), type=type(frame).__name__,
                        warnings=[dict(category=type(w.message).__name__,
                                       message=str(w.message)) for w in caught]))
(output / os.environ['MALAWI_MANIFEST']).write_text(json.dumps(results, indent=2))
#+end_src
